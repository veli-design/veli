## Create a project

Use Node.js 24 or newer. The default template is framework mode, with routes in `app/routes` and the document shell in `app/root.tsx`.

```bash
npx create-react-router@latest my-veli-app
cd my-veli-app
npm install
```

## Install packages

Install the shared packages from [Installation](/getting-started/installation/).

```bash setup=install
npm install @veli-design/react @veli-design/tokens react-aria-components
```

If the template did not add Tailwind 4, install the Vite plugin.

```bash
npm install tailwindcss@^4 @tailwindcss/vite@^4
```

## Add styles

Add `tailwindcss()` to the existing Vite plugins. Keep `reactRouter()` and `tsconfigPaths()`.

```ts
import { reactRouter } from '@react-router/dev/vite';
import { defineConfig } from 'vite';
import tsconfigPaths from 'vite-tsconfig-paths';
import tailwindcss from '@tailwindcss/vite';

export default defineConfig({
  plugins: [tailwindcss(), reactRouter(), tsconfigPaths()],
});
```

Put the Veli imports at the top of `app/app.css`.

```css setup=css-imports
@import 'tailwindcss';
@import '@veli-design/tokens/tailwind';
@import '@veli-design/react/tailwind';
```

## Add the provider

`use client` marks this module as a client component. The saved color scheme is read after mount so a server render and the first client render agree.

```tsx setup=client-theme
'use client';

import { useEffect, useState } from 'react';
import { VeliProvider, type VeliColorScheme } from '@veli-design/react/provider';

const storageKey = 'veli-color-scheme';

function isColorScheme(value: string | null): value is VeliColorScheme {
  return value === 'light' || value === 'dark' || value === 'system';
}

function resolvedScheme(colorScheme: VeliColorScheme) {
  if (colorScheme === 'light' || colorScheme === 'dark') return colorScheme;
  return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
}

export function AppThemeProvider({ children }: { children: React.ReactNode }) {
  const [colorScheme, setColorScheme] = useState<VeliColorScheme>('system');

  useEffect(() => {
    const stored = window.localStorage.getItem(storageKey);
    if (isColorScheme(stored) && stored !== colorScheme) setColorScheme(stored);
  }, [colorScheme]);

  useEffect(() => {
    const resolved = resolvedScheme(colorScheme);
    const root = document.documentElement;
    root.classList.toggle('dark', resolved === 'dark');
    root.dataset.theme = resolved;
    root.dataset.colorScheme = colorScheme;
    root.style.colorScheme = resolved;
    window.localStorage.setItem(storageKey, colorScheme);

    const media = window.matchMedia('(prefers-color-scheme: dark)');
    const onChange = () => {
      if (root.dataset.colorScheme !== 'system') return;
      const next = media.matches ? 'dark' : 'light';
      root.classList.toggle('dark', next === 'dark');
      root.dataset.theme = next;
      root.style.colorScheme = next;
    };
    media.addEventListener('change', onChange);
    return () => media.removeEventListener('change', onChange);
  }, [colorScheme]);

  return (
    <VeliProvider colorScheme={colorScheme}>
      <button type="button" onClick={() => setColorScheme('system')}>
        System
      </button>
      <button type="button" onClick={() => setColorScheme('light')}>
        Light
      </button>
      <button type="button" onClick={() => setColorScheme('dark')}>
        Dark
      </button>
      {children}
    </VeliProvider>
  );
}
```

Wrap `<Outlet />` in `app/root.tsx` with `AppThemeProvider`.

## Connect the router

Pass `navigate` and `useHref` so links from Veli components stay inside the router. Call this component inside `AppThemeProvider`, or merge the two wrappers.

```tsx
'use client';

import { useNavigate, useHref } from 'react-router';
import { VeliProvider } from '@veli-design/react/provider';

export function AppProviders({ children }: { children: React.ReactNode }) {
  const navigate = useNavigate();

  return (
    <VeliProvider navigate={navigate} useHref={useHref}>
      {children}
    </VeliProvider>
  );
}
```

If you already wrap the tree in `AppThemeProvider`, pass `navigate` and `useHref` to that `VeliProvider` instead of mounting a second provider.

## Load fonts

Name the Latin face Inter and `--veli-font-sans` selects it. [Fonts](/getting-started/fonts/) has the host rules for Inter and the Dhivehi faces. Import the font stylesheet from `app/root.tsx` or from `app/app.css`.

## Verify

```bash
npm run dev
```

Open a route that renders a Veli button. The button uses Veli styles. Choose **Dark**. The document element gains the class `dark`. Activate a Veli link that has an `href`. The address changes without a full page load.
