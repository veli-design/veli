## Create a project

Use Node.js 24 or newer. Choose TypeScript, Tailwind CSS, and the App Router.

```bash
npx create-next-app@latest my-veli-app
cd my-veli-app
```

## Install packages

Install the shared packages from [Installation](/getting-started/installation/). The App Router template already includes Tailwind.

```bash setup=install
npm install @veli-design/react @veli-design/tokens react-aria-components
```

Remove `@veli-design/react` from `transpilePackages` if an older release added it. Current packages ship explicit `.js` imports and do not need that workaround.

## Add styles

Keep the Tailwind import the template generated. Add the two Veli imports after it, in `app/globals.css`.

```css setup=css-imports
@import 'tailwindcss';
@import '@veli-design/tokens/tailwind';
@import '@veli-design/react/tailwind';
```

If the template's Tailwind import uses double quotes, change that line to the single-quoted import above so the three lines stay in this order.

## Add the provider

Use `next-themes` when the app already toggles `class="dark"` on `html`. Set `attribute="class"`. `VeliProvider` can still supply language and component context. Do not also let it own the document theme.

```tsx
import { ThemeProvider } from 'next-themes';

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body>
        <ThemeProvider
          attribute="class"
          defaultTheme="system"
          enableSystem
          disableTransitionOnChange
        >
          {children}
        </ThemeProvider>
      </body>
    </html>
  );
}
```

Bridge the theme into `VeliProvider` from a client module.

```tsx
'use client';

import { useTheme } from 'next-themes';
import { VeliProvider, type VeliColorScheme } from '@veli-design/react/provider';

function toVeliColorScheme(theme: string | undefined): VeliColorScheme {
  return theme === 'light' || theme === 'dark' ? theme : 'system';
}

export function VeliThemeBridge({ children }: { children: React.ReactNode }) {
  const { theme } = useTheme();

  return <VeliProvider colorScheme={toVeliColorScheme(theme)}>{children}</VeliProvider>;
}
```

Render `VeliThemeBridge` inside `ThemeProvider`. Interactive Veli components are client components. Import them from client modules.

## Connect the router

Pass `navigate` so `href` on Link, Tabs, Menu, and Breadcrumbs uses the App Router.

```tsx
'use client';

import { useRouter } from 'next/navigation';
import { VeliProvider } from '@veli-design/react/provider';

export function AppProviders({ children }: { children: React.ReactNode }) {
  const router = useRouter();

  return (
    <VeliProvider navigate={(href, options) => router.push(href, options)}>{children}</VeliProvider>
  );
}
```

To type `routerOptions` on individual links, augment React Aria's `RouterConfig`.

```ts
import type { useRouter } from 'next/navigation';

declare module 'react-aria-components' {
  interface RouterConfig {
    routerOptions: NonNullable<Parameters<ReturnType<typeof useRouter>['push']>[1]>;
  }
}
```

`navigate` is optional. Without it, `href` uses a full page load.

## Load fonts

Bind the loader's CSS variable to `--veli-font-sans` and put the generated class on `html`. [Fonts](/getting-started/fonts/) has the App Router example and the Dhivehi family rules. Import the global stylesheet from the root layout.

## Verify

```bash
npm run dev
```

Open the app. A Veli button uses Veli styles. Switch the theme. The document element gains or loses the class `dark`. Activate a Veli link that has an `href`. The App Router changes the address without a full page load.
